{{- /* Prefer editorial descriptions; generate missing ones at build time, including historical releases, so summaries cannot become stale copies of release data. */ -}} {{- $page := .page -}} {{- $version := $page.File.BaseFileName -}} {{- $release := index site.Data.docs.releases $version -}} {{- $description := "" -}} {{- with $release.description -}} {{- $description = . | markdownify -}} {{- else -}} {{- with $release.changelog -}} {{- $changelog := . -}} {{- $contributors := slice -}} {{- range $changelog -}} {{- range .contributors -}} {{- $contributors = $contributors | append . -}} {{- end -}} {{- end -}} {{- $contributors = $contributors | uniq -}} {{- $highlights := where (where site.RegularPages "Section" "highlights") ".Params.release" "eq" $version -}} {{- $guides := where $highlights ".Title" "like" "Upgrade Guide$" -}} {{- $upgradeURL := "" -}} {{- if and $page.Params.inline_upgrade_guide (ne (strings.TrimSpace $page.Content) "") -}} {{- $upgradeURL = printf "%s#upgrade-guide" $page.RelPermalink -}} {{- else -}} {{- with $guides -}} {{- $upgradeURL = (index . 0).RelPermalink -}} {{- end -}} {{- end -}} {{- /* Link only to sections the release page actually renders. */ -}} {{- $breaking := slice -}} {{- $grouped := slice -}} {{- range $changelog -}} {{- if and .breaking (ne .title nil) $upgradeURL -}} {{- $breaking = $breaking | append . -}} {{- else -}} {{- $grouped = $grouped | append . -}} {{- end -}} {{- end -}} {{- $sections := slice -}} {{- with $breaking -}} {{- $sections = $sections | append (dict "count" (len .) "singular" "breaking change" "plural" "breaking changes" "anchor" "breaking-changes") -}} {{- end -}} {{- range slice (dict "type" "security" "singular" "security change" "plural" "security changes" "anchor" "security-changes") (dict "type" "feat" "singular" "new feature" "plural" "new features" "anchor" "new-features") (dict "type" "enhancement" "singular" "enhancement" "plural" "enhancements" "anchor" "enhancements") (dict "type" "fix" "singular" "bug fix" "plural" "bug fixes" "anchor" "bug-fixes") (dict "type" "deprecation" "singular" "deprecation" "plural" "deprecations" "anchor" "deprecations") (dict "type" "chore" "singular" "chore" "plural" "chores" "anchor" "chore") -}} {{- $count := len (where $grouped "type" .type) -}} {{- if $count -}} {{- $sections = $sections | append (merge . (dict "count" $count)) -}} {{- end -}} {{- end -}} {{- $description = printf "The [COSE team](https://opensource.datadoghq.com/about/#the-community-open-source-engineering-team) is happy to announce Vector `%s`!" $version -}} {{- $changes := printf "%d changes" (len $changelog) -}} {{- if eq (len $changelog) 1 -}} {{- $changes = "change" -}} {{- end -}} {{- if $contributors -}} {{- $people := printf "%d contributors" (len $contributors) -}} {{- if eq (len $contributors) 1 -}} {{- $people = "contributor" -}} {{- end -}} {{- $description = printf "%s\n\nThanks to the %s who made the %s in this release:" $description $people $changes -}} {{- else -}} {{- $description = printf "%s\n\nThis release includes %d %s:" $description (len $changelog) (cond (eq (len $changelog) 1) "change" "changes") -}} {{- end -}} {{- $description = printf "%s\n" $description -}} {{- range $sections -}} {{- $noun := cond (eq .count 1) .singular .plural -}} {{- $description = printf "%s\n- [%d %s](%s#%s)" $description .count $noun $page.RelPermalink .anchor -}} {{- end -}} {{- with $upgradeURL -}} {{- $description = printf "%s\n\nUpgrading? Read the [upgrade guide](%s)." $description . -}} {{- end -}} {{- if $.listing -}} {{- $description = printf "%s\n\n[View the full release notes](%s)." $description $page.RelPermalink -}} {{- end -}} {{- $description = $description | markdownify -}} {{- end -}} {{- end -}} {{- return $description -}}
