{{- /* Returns the deprecation notices announced in a specific release: active notices first, then notices that have since been enacted (these carry `removed_in`). Enacted notices stay listed on the release that announced them; the release that removes them surfaces the removal in its Breaking changes section instead. Params: version (string, e.g. "0.56" or "0.56.0") */ -}} {{- $entries := partialCached "deprecations/entries.html" . -}} {{- /* Entries carry X.Y.Z versions (see deprecations/entries.html), so normalize a "0.56" shorthand param the same way and match exactly: "0.56" fragments belong to the 0.56.0 page only, never to 0.56.1. */ -}} {{- $version := (partial "deprecations/version.html" .version).version -}} {{- $active := where $entries.active "deprecated_since" $version -}} {{- $enacted := where $entries.enacted "deprecated_since" $version -}} {{- return ($active | append $enacted) -}}
